vec2 length2(vec4 a) {

vec2 RoundedRectangle(vec2 dim, float r, float bw) {              
  dim -= bw +.0002; // Subtract the border                        
  r = min(r, min(dim.x, dim.y));                                  
  dim -= r; // Subtract the border-radius                         
  vec4 d = abs(pos) - dim.xyxy;                                   
  vec4 dmin = min(d, 0.0);                                        
  vec4 dmax = max(d, 0.0);                                        
  vec2 df = max(dmin.xz, dmin.yw) + length2(dmax);                
  return (df - r);                                                
}
